3. The important modern Node/bencode change

Older tutorials may use torrent.announce.toString('utf8'). With bencode@4, decoded byte strings are often Uint8Array values. Calling .toString() directly on a Uint8Array gives comma-separated numbers, not text.

// Wrong with bencode@4 decoded Uint8Array values:
torrent.announce.toString('utf8')

// Correct:
Buffer.from(torrent.announce).toString('utf8')

4. Create src/torrent-parser.js

This file reads, decodes, and normalises the torrent metadata. It does not download anything yet.

import fs from 'node:fs';
import crypto from 'node:crypto';
import bencode from 'bencode';
import { Buffer } from 'node:buffer';

export function openTorrent(torrentPath) {
  const torrentFile = fs.readFileSync(torrentPath);
  return bencode.decode(torrentFile);
}

export function bytesToString(value) {
  if (value === undefined || value === null) return '';
  return Buffer.from(value).toString('utf8');
}

export function bytesToHex(value) {
  return Buffer.from(value).toString('hex');
}

export function size(torrent) {
  const info = torrent.info;

  if (info.length !== undefined) {
    return Number(info.length);
  }

  if (Array.isArray(info.files)) {
    return info.files.reduce((total, file) => total + Number(file.length), 0);
  }

  throw new Error('Invalid torrent: expected info.length or info.files');
}

export function torrentName(torrent) {
  return bytesToString(torrent.info.name);
}

export function announceUrl(torrent) {
  return bytesToString(torrent.announce);
}

export function infoHash(torrent) {
  const encodedInfo = bencode.encode(torrent.info);
  return crypto.createHash('sha1').update(encodedInfo).digest();
}

export function infoHashHex(torrent) {
  return infoHash(torrent).toString('hex');
}

export function pieceLength(torrent) {
  return Number(torrent.info['piece length']);
}

export function pieces(torrent) {
  const piecesBuffer = Buffer.from(torrent.info.pieces);
  const hashes = [];

  for (let offset = 0; offset < piecesBuffer.length; offset += 20) {
    hashes.push(piecesBuffer.subarray(offset, offset + 20));
  }

  return hashes;
}

export function files(torrent) {
  const info = torrent.info;

  if (info.length !== undefined) {
    return [
      {
        path: torrentName(torrent),
        length: Number(info.length)
      }
    ];
  }

  return info.files.map((file) => ({
    path: file.path.map(bytesToString).join('/'),
    length: Number(file.length)
  }));
}

function percentEncodeBytes(value) {
  return Array.from(Buffer.from(value))
    .map((byte) => `%${byte.toString(16).padStart(2, '0')}`)
    .join('');
}

export function buildHttpTrackerUrl(torrent, peerId, port = 6881) {
  const announce = announceUrl(torrent);

  if (!announce.startsWith('http://') && !announce.startsWith('https://')) {
    throw new Error(`This helper only supports HTTP/HTTPS trackers. Tracker was: ${announce}`);
  }

  const params = [
    ['info_hash', percentEncodeBytes(infoHash(torrent))],
    ['peer_id', percentEncodeBytes(peerId)],
    ['port', String(port)],
    ['uploaded', '0'],
    ['downloaded', '0'],
    ['left', String(size(torrent))],
    ['compact', '1'],
    ['event', 'started']
  ];

  const query = params.map(([key, value]) => `${key}=${value}`).join('&');
  const separator = announce.includes('?') ? '&' : '?';

  return `${announce}${separator}${query}`;
}

export function createPeerId() {
  const prefix = Buffer.from('-AT0001-');
  const random = crypto.randomBytes(12);
  return Buffer.concat([prefix, random]);
}

5. Create src/index.js

This is the command-line entry point. It prints torrent metadata and the first few piece hashes.

import {
  openTorrent,
  announceUrl,
  torrentName,
  size,
  pieceLength,
  pieces,
  files,
  infoHashHex,
  createPeerId,
  buildHttpTrackerUrl
} from './torrent-parser.js';

const torrentPath = process.argv[2];

if (!torrentPath) {
  console.error('Usage: npm run parse -- path/to/file.torrent');
  process.exit(1);
}

const torrent = openTorrent(torrentPath);
const pieceHashes = pieces(torrent);

console.log('Torrent metadata');
console.log('----------------');
console.log('Name:          ', torrentName(torrent));
console.log('Announce:      ', announceUrl(torrent));
console.log('Total size:    ', `${size(torrent).toLocaleString()} bytes`);
console.log('Piece length:  ', `${pieceLength(torrent).toLocaleString()} bytes`);
console.log('Piece count:   ', pieceHashes.length);
console.log('Info hash hex: ', infoHashHex(torrent));

console.log('\nFiles');
console.log('-----');
for (const file of files(torrent)) {
  console.log(`${file.length.toLocaleString().padStart(12)} bytes  ${file.path}`);
}

console.log('\nFirst 5 piece hashes');
console.log('--------------------');
for (const hash of pieceHashes.slice(0, 5)) {
  console.log(hash.toString('hex'));
}

try {
  const peerId = createPeerId();
  const trackerUrl = buildHttpTrackerUrl(torrent, peerId);
  console.log('\nExample HTTP tracker URL');
  console.log('------------------------');
  console.log(trackerUrl.toString());
} catch (error) {
  console.log('\nTracker URL note');
  console.log('----------------');
  console.log(error.message);
}

If your tracker starts with udp://, the metadata parser still works. UDP trackers use a different binary protocol.

7. How the code works

openTorrent()

Reads the file as raw bytes and decodes bencode into a JavaScript object.

infoHash()

Hashes only the bencoded info dictionary. Do not hash JSON or the full file.

pieces()

Splits the long pieces byte string into 20-byte SHA-1 hashes.

files()

Returns one consistent array for both single-file and multi-file torrents.

8. Troubleshooting common errors

bencode does not provide an export named decode
import bencode from 'bencode';
const torrent = bencode.decode(data);
announce prints numbers instead of a URL
Buffer.from(torrent.announce).toString('utf8')
require is not defined in ES module scope
import fs from 'node:fs';
Cannot find module src/torrent-parser.js

Check folder names and include the .js extension in your import path.

9. What to build next

  1. HTTP tracker request: use fetch(trackerUrl), decode the response, and parse compact peers.
  2. UDP tracker support: implement BEP 15 connect and announce packets with dgram.
  3. Peer handshake: open a TCP socket and send protocol string, reserved bytes, info_hash, and peer_id.
  4. Peer messages: parse choke, unchoke, interested, bitfield, request, and piece messages.
  5. Piece verification: SHA-1 hash downloaded pieces and compare them against pieces().

10. Optional: HTTP tracker fetch skeleton

Use this only for HTTP/HTTPS trackers. It will not work for udp:// trackers.

import bencode from 'bencode';
import { Buffer } from 'node:buffer';
import { openTorrent, createPeerId, buildHttpTrackerUrl } from './torrent-parser.js';

function parseCompactPeers(peersValue) {
  const peers = Buffer.from(peersValue);
  const result = [];

  for (let i = 0; i < peers.length; i += 6) {
    const ip = `${peers[i]}.${peers[i + 1]}.${peers[i + 2]}.${peers[i + 3]}`;
    const port = peers.readUInt16BE(i + 4);
    result.push({ ip, port });
  }

  return result;
}

const torrent = openTorrent(process.argv[2]);
const peerId = createPeerId();
const trackerUrl = buildHttpTrackerUrl(torrent, peerId);

const response = await fetch(trackerUrl);
const responseBytes = new Uint8Array(await response.arrayBuffer());
const trackerResponse = bencode.decode(responseBytes);

if (trackerResponse['failure reason']) {
  throw new Error(Buffer.from(trackerResponse['failure reason']).toString('utf8'));
}

console.log('Interval:', trackerResponse.interval);
console.log('Peers:', parseCompactPeers(trackerResponse.peers));

11. Final checklist

  • Use "type": "module" in package.json.
  • Use import syntax, not require.
  • Import bencode as default: import bencode from 'bencode'.
  • Convert bencode byte strings with Buffer.from(value).toString('utf8').
  • Hash bencode.encode(torrent.info), not the whole file.
  • Split torrent.info.pieces into 20-byte chunks.
  • Handle single-file and multi-file torrents separately.
  • Treat udp:// trackers separately from HTTP trackers.
