8.2 Generate one peer ID for the process

src/util.js

JavaScript
import { Buffer } from 'node:buffer';
import crypto from 'node:crypto';

const PEER_ID_PREFIX = Buffer.from('-AT0002-', 'ascii');
let peerId;

export function getPeerId() {
  if (!peerId) {
    peerId = Buffer.concat([PEER_ID_PREFIX, crypto.randomBytes(12)]);
  }

  return peerId;
}

The prefix is eight bytes and the random suffix is twelve bytes, producing exactly twenty bytes. The same peer ID is reused for tracker and peer messages during one program run. AT is simply an identifier chosen for this tutorial; it is not an officially registered client code.

8.3 Calculate the info hash

20-byte v1 torrent identifier

JavaScript
export function infoHash(torrent) {
  const encodedInfo = bencode.encode(torrent.info);
  return crypto.createHash('sha1').update(encodedInfo).digest();
}

digest without an encoding argument returns a Buffer. Do not ask for a hex string here, because the protocol requires the raw 20 bytes, not 40 ASCII hex characters.

8.4 Calculate torrent size with BigInt

Single-file and multi-file size calculation

JavaScript
export function totalSizeBigInt(torrent) {
  if (torrent.info.files) {
    return torrent.info.files.reduce(
      (total, file) => total + BigInt(file.length),
      0n,
    );
  }

  return BigInt(torrent.info.length);
}

The final program rejects multi-file downloads, but calculating their total logical size remains useful and preserves the original tutorial's explanation. 0n is a BigInt zero. Every file length is converted to BigInt before addition.

8.5 Build the announce request

Announce request builder

JavaScript
export function buildAnnounceRequest(connectionId, torrent, port = 6881) {
  if (connectionId.length !== 8) {
    throw new Error('A UDP tracker connection ID must contain 8 bytes.');
  }
 
  const transactionId = randomUInt32();
  const buffer = Buffer.alloc(98);
 
  connectionId.copy(buffer, 0);
  buffer.writeUInt32BE(1, 8);
  buffer.writeUInt32BE(transactionId, 12);
  infoHash(torrent).copy(buffer, 16);
  getPeerId().copy(buffer, 36);
  buffer.writeBigUInt64BE(0n, 56);
  buffer.writeBigUInt64BE(totalSizeBigInt(torrent), 64);
  buffer.writeBigUInt64BE(0n, 72);
  buffer.writeUInt32BE(2, 80);
  buffer.writeUInt32BE(0, 84);
  buffer.writeUInt32BE(randomUInt32(), 88);
  buffer.writeInt32BE(-1, 92);
  buffer.writeUInt16BE(port, 96);
 
  return { buffer, transactionId };
}

The older article parser read leechers at offset 8 and seeders at 12, skipping interval and shifting the meanings. The corrected parser follows BEP 15.

8.7 Parse compact peers

Correct announce response parser

JavaScript
export function parseAnnounceResponse(response, expectedTransactionId) {
  ensureTrackerResponse(response, 20, 1, expectedTransactionId);

  if ((response.length - 20) % 6 !== 0) {
    throw new Error('Tracker returned a malformed compact peer list.');
  }

  const peers = [];

  for (let offset = 20; offset < response.length; offset += 6) {
    peers.push({
      ip: [
        response[offset],
        response[offset + 1],
        response[offset + 2],
        response[offset + 3],
      ].join('.'),
      port: response.readUInt16BE(offset + 4),
    });
  }

  return {
    action: response.readUInt32BE(0),
    transactionId: response.readUInt32BE(4),
    interval: response.readUInt32BE(8),
    leechers: response.readUInt32BE(12),
    seeders: response.readUInt32BE(16),
    peers,
  };
}

8.8 Send, timeout, retry, and match

sendOnce attaches temporary message and error listeners, starts a timer, sends the packet, and resolves only when a response has the correct transaction ID and an expected action. Cleanup removes listeners and the timer so later exchanges do not accidentally process the same packet.

Exponential timeout sequence

JavaScript
async function sendWithRetries(options) {
  let lastError;

  for (let attempt = 0; attempt < options.retries; attempt += 1) {
    const timeoutMs = 15_000 * (2 ** attempt);

    try {
      return await sendOnce({ ...options, timeoutMs });
    } catch (error) {
      lastError = error;
      if (error.code !== 'TRACKER_TIMEOUT') throw error;
    }
  }

  throw lastError;
}

8.9 The complete getPeers flow

Readable high-level outline

JavaScript
export async function getPeers(torrent, options = {}) {
  const trackerUrl = new URL(chooseUdpTracker(torrent));
  const port = Number(trackerUrl.port);
  const socket = dgram.createSocket('udp4');

  try {
    const connectRequest = buildConnectRequest();
    const connectBytes = await sendWithRetries({ /* connect details */ });
    const connect = parseConnectResponse(
      connectBytes,
      connectRequest.transactionId,
    );

    const announceRequest = buildAnnounceRequest(
      connect.connectionId,
      torrent,
      options.listenPort ?? 6881,
    );
    const announceBytes = await sendWithRetries({ /* announce details */ });

    return parseAnnounceResponse(
      announceBytes,
      announceRequest.transactionId,
    ).peers;
  } finally {
    socket.close();
  }
}
