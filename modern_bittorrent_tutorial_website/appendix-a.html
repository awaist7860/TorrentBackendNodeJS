JavaScript
import { Buffer } from 'node:buffer';
import crypto from 'node:crypto';
import fs from 'node:fs';
 
import bencode from 'bencode';
 
export const BLOCK_LENGTH = 16 * 1024;
 
export function openTorrent(filePath) {
  const file = fs.readFileSync(filePath);
  return bencode.decode(file);
}
 
export function isSingleFileTorrent(torrent) {
  return torrent.info.files === undefined;
}
 
export function torrentName(torrent) {
  return Buffer.from(torrent.info.name).toString('utf8');
}
 
export function trackerUrls(torrent) {
  const urls = [];
 
  if (torrent.announce) {
    urls.push(Buffer.from(torrent.announce).toString('utf8'));
  }
 
  if (Array.isArray(torrent['announce-list'])) {
    for (const tier of torrent['announce-list']) {
      const entries = Array.isArray(tier) ? tier : [tier];
      for (const entry of entries) {
        urls.push(Buffer.from(entry).toString('utf8'));
      }
    }
  }
 
  return [...new Set(urls)];
}
 
export function chooseUdpTracker(torrent) {
  const url = trackerUrls(torrent).find((candidate) =>
    candidate.startsWith('udp://'),
  );
 
  if (!url) {
    throw new Error(
      'No UDP tracker was found in this torrent. ' +
      'This learning client does not yet support HTTP trackers or DHT.',
    );
  }
 
  return url;
}
 
export function infoHash(torrent) {
  const encodedInfo = bencode.encode(torrent.info);
  return crypto.createHash('sha1').update(encodedInfo).digest();
}
 
export function totalSizeBigInt(torrent) {
  if (torrent.info.files) {
    return torrent.info.files.reduce(
      (total, file) => total + BigInt(file.length),
      0n,
    );
  }
 
  return BigInt(torrent.info.length);
}
 
export function totalSizeBuffer(torrent) {
  const buffer = Buffer.alloc(8);
  buffer.writeBigUInt64BE(totalSizeBigInt(torrent));
  return buffer;
}
 
export function numberOfPieces(torrent) {
  const count = torrent.info.pieces.length / 20;
 
  if (!Number.isInteger(count)) {
    throw new Error('The torrent pieces field is not a multiple of 20 bytes.');
  }
 
  return count;
}
 
export function pieceLength(torrent, pieceIndex) {
  const pieceCount = numberOfPieces(torrent);
 
  if (pieceIndex < 0 || pieceIndex >= pieceCount) {
    throw new RangeError(`Invalid piece index: ${pieceIndex}`);
  }
 
  const normalLength = BigInt(torrent.info['piece length']);
 
  if (pieceIndex < pieceCount - 1) {
    return Number(normalLength);
  }
 
  const bytesBeforeLastPiece = normalLength * BigInt(pieceCount - 1);
  return Number(totalSizeBigInt(torrent) - bytesBeforeLastPiece);
}
 
export function blocksPerPiece(torrent, pieceIndex) {
  return Math.ceil(pieceLength(torrent, pieceIndex) / BLOCK_LENGTH);
}
 
export function blockLength(torrent, pieceIndex, blockIndex) {
  const begin = blockIndex * BLOCK_LENGTH;
  const remaining = pieceLength(torrent, pieceIndex) - begin;
 
  if (remaining <= 0) {
    throw new RangeError(
      `Invalid block index ${blockIndex} for piece ${pieceIndex}`,
    );
  }
 
  return Math.min(BLOCK_LENGTH, remaining);
}
 
export function expectedPieceHash(torrent, pieceIndex) {
  const start = pieceIndex * 20;
  return Buffer.from(torrent.info.pieces).subarray(start, start + 20);
}

src/message.js

JavaScript
import { Buffer } from 'node:buffer';
import { infoHash } from './torrent-parser.js';
import { getPeerId } from './util.js';
 
const PROTOCOL_NAME = 'BitTorrent protocol';
const PROTOCOL_NAME_BUFFER = Buffer.from(PROTOCOL_NAME, 'utf8');
 
export function buildHandshake(torrent) {
  const buffer = Buffer.alloc(68);
 
  buffer.writeUInt8(PROTOCOL_NAME_BUFFER.length, 0);
  PROTOCOL_NAME_BUFFER.copy(buffer, 1);
  // Bytes 20-27 are reserved extension bits. Buffer.alloc() sets them to zero.
  infoHash(torrent).copy(buffer, 28);
  getPeerId().copy(buffer, 48);
 
  return buffer;
}
 
export function parseHandshake(buffer) {
  if (buffer.length < 49) {
    throw new Error('Handshake is too short.');
  }
 
  const protocolLength = buffer.readUInt8(0);
  const expectedLength = 49 + protocolLength;
 
  if (buffer.length !== expectedLength) {
    throw new Error('Handshake has an invalid length.');
  }
 
  const protocol = buffer.toString('utf8', 1, 1 + protocolLength);
  const infoHashStart = 1 + protocolLength + 8;
 
  return {
    protocol,
    infoHash: buffer.subarray(infoHashStart, infoHashStart + 20),
    peerId: buffer.subarray(infoHashStart + 20, infoHashStart + 40),
  };
}
 
export function buildKeepAlive() {
  return Buffer.alloc(4);
}
 
function buildMessage(id, payload = Buffer.alloc(0)) {
  const buffer = Buffer.alloc(4 + 1 + payload.length);
  buffer.writeUInt32BE(1 + payload.length, 0);
  buffer.writeUInt8(id, 4);
  payload.copy(buffer, 5);
  return buffer;
}
 
export function buildChoke() {
  return buildMessage(0);
}
 
export function buildUnchoke() {
  return buildMessage(1);
}
 
export function buildInterested() {
  return buildMessage(2);
}
 
export function buildUninterested() {
  return buildMessage(3);
}
 
export function buildHave(pieceIndex) {
  const payload = Buffer.alloc(4);
  payload.writeUInt32BE(pieceIndex, 0);
  return buildMessage(4, payload);
}
 
export function buildBitfield(bitfield) {
  return buildMessage(5, bitfield);
}
 
export function buildRequest({ index, begin, length }) {
  const payload = Buffer.alloc(12);
  payload.writeUInt32BE(index, 0);
  payload.writeUInt32BE(begin, 4);
  payload.writeUInt32BE(length, 8);
  return buildMessage(6, payload);
}
 
export function buildPiece({ index, begin, block }) {
  const payload = Buffer.alloc(8 + block.length);
  payload.writeUInt32BE(index, 0);
  payload.writeUInt32BE(begin, 4);
  block.copy(payload, 8);
  return buildMessage(7, payload);
}
 
export function buildCancel({ index, begin, length }) {
  const payload = Buffer.alloc(12);
  payload.writeUInt32BE(index, 0);
  payload.writeUInt32BE(begin, 4);
  payload.writeUInt32BE(length, 8);
  return buildMessage(8, payload);
}
 
export function buildPort(port) {
  const payload = Buffer.alloc(2);
  payload.writeUInt16BE(port, 0);
  return buildMessage(9, payload);
}
 
export function parseMessage(message) {
  if (message.length < 4) {
    throw new Error('Peer message is too short.');
  }
 
  const size = message.readUInt32BE(0);
 
  if (message.length !== size + 4) {
    throw new Error('Peer message length prefix does not match its data.');
  }
 
  if (size === 0) {
    return { size, id: null, payload: null };
  }
 
  const id = message.readUInt8(4);
  const rawPayload = message.subarray(5);
  let payload = rawPayload;
 
  if (id === 6 || id === 8) {
    if (rawPayload.length !== 12) {
      throw new Error('Request/cancel message has an invalid payload length.');
    }
 
    payload = {
      index: rawPayload.readUInt32BE(0),
      begin: rawPayload.readUInt32BE(4),
      length: rawPayload.readUInt32BE(8),
    };
  } else if (id === 7) {
    if (rawPayload.length < 8) {
      throw new Error('Piece message has an invalid payload length.');
    }
 
    payload = {
      index: rawPayload.readUInt32BE(0),
      begin: rawPayload.readUInt32BE(4),
      block: rawPayload.subarray(8),
    };
  }
 
  return { size, id, payload };
}

src/tracker.js

JavaScript
import { Buffer } from 'node:buffer';
import dgram from 'node:dgram';
 
import {
  chooseUdpTracker,
  infoHash,
  totalSizeBigInt,
} from './torrent-parser.js';
import { getPeerId, randomUInt32 } from './util.js';
 
const UDP_TRACKER_PROTOCOL_ID = 0x41727101980n;
const ACTION_CONNECT = 0;
const ACTION_ANNOUNCE = 1;
const ACTION_ERROR = 3;
 
export async function getPeers(torrent, options = {}) {
  const trackerUrl = new URL(chooseUdpTracker(torrent));
  const port = Number(trackerUrl.port);
  const listenPort = options.listenPort ?? 6881;
  const retries = options.retries ?? 3;
 
  if (!port) {
    throw new Error('The UDP tracker URL does not contain a port number.');
  }
 
  const socket = dgram.createSocket('udp4');
 
  try {
    const connectRequest = buildConnectRequest();
    const connectResponseBuffer = await sendWithRetries({
      socket,
      trackerUrl,
      port,
      message: connectRequest.buffer,
      transactionId: connectRequest.transactionId,
      expectedAction: ACTION_CONNECT,
      retries,
    });
 
    const connectResponse = parseConnectResponse(
      connectResponseBuffer,
      connectRequest.transactionId,
    );
 
    const announceRequest = buildAnnounceRequest(
      connectResponse.connectionId,
      torrent,
      listenPort,
    );
 
    const announceResponseBuffer = await sendWithRetries({
      socket,
      trackerUrl,
      port,
      message: announceRequest.buffer,
      transactionId: announceRequest.transactionId,
      expectedAction: ACTION_ANNOUNCE,
      retries,
    });
 
    return parseAnnounceResponse(
      announceResponseBuffer,
      announceRequest.transactionId,
    ).peers;
  } finally {
    socket.close();
  }
}
 
export function buildConnectRequest() {
  const transactionId = randomUInt32();
  const buffer = Buffer.alloc(16);
 
  buffer.writeBigUInt64BE(UDP_TRACKER_PROTOCOL_ID, 0);
  buffer.writeUInt32BE(ACTION_CONNECT, 8);
  buffer.writeUInt32BE(transactionId, 12);
 
  return { buffer, transactionId };
}
 
export function parseConnectResponse(response, expectedTransactionId) {
  ensureTrackerResponse(response, 16, ACTION_CONNECT, expectedTransactionId);
 
  return {
    action: response.readUInt32BE(0),
    transactionId: response.readUInt32BE(4),
    connectionId: response.subarray(8, 16),
  };
}
 
export function buildAnnounceRequest(connectionId, torrent, port = 6881) {
  if (connectionId.length !== 8) {
    throw new Error('A UDP tracker connection ID must contain 8 bytes.');
  }
 
  const transactionId = randomUInt32();
  const buffer = Buffer.alloc(98);
 
  connectionId.copy(buffer, 0);
  buffer.writeUInt32BE(ACTION_ANNOUNCE, 8);
  buffer.writeUInt32BE(transactionId, 12);
  infoHash(torrent).copy(buffer, 16);
  getPeerId().copy(buffer, 36);
  buffer.writeBigUInt64BE(0n, 56); // downloaded
  buffer.writeBigUInt64BE(totalSizeBigInt(torrent), 64); // left
  buffer.writeBigUInt64BE(0n, 72); // uploaded
  buffer.writeUInt32BE(2, 80); // event: started
  buffer.writeUInt32BE(0, 84); // IP address: use the sender's address
  buffer.writeUInt32BE(randomUInt32(), 88); // key
  buffer.writeInt32BE(-1, 92); // number wanted: tracker default
  buffer.writeUInt16BE(port, 96);
 
  return { buffer, transactionId };
}
 
export function parseAnnounceResponse(response, expectedTransactionId) {
  ensureTrackerResponse(response, 20, ACTION_ANNOUNCE, expectedTransactionId);
 
  if ((response.length - 20) % 6 !== 0) {
    throw new Error('Tracker returned a malformed compact peer list.');
  }
 
  const peers = [];
 
  for (let offset = 20; offset < response.length; offset += 6) {
    peers.push({
      ip: [
        response[offset],
        response[offset + 1],
        response[offset + 2],
        response[offset + 3],
      ].join('.'),
      port: response.readUInt16BE(offset + 4),
    });
  }
 
  return {
    action: response.readUInt32BE(0),
    transactionId: response.readUInt32BE(4),
    interval: response.readUInt32BE(8),
    leechers: response.readUInt32BE(12),
    seeders: response.readUInt32BE(16),
    peers,
  };
}
 
function ensureTrackerResponse(
  response,
  minimumLength,
  expectedAction,
  expectedTransactionId,
) {
  if (response.length < 8) {
    throw new Error('Tracker response is too short.');
  }
 
  const action = response.readUInt32BE(0);
  const transactionId = response.readUInt32BE(4);
 
  if (transactionId !== expectedTransactionId) {
    throw new Error('Tracker response has the wrong transaction ID.');
  }
 
  if (action === ACTION_ERROR) {
    const message = response.subarray(8).toString('utf8');
    throw new Error(`Tracker error: ${message}`);
  }
 
  if (action !== expectedAction) {
    throw new Error(`Unexpected tracker action: ${action}`);
  }
 
  if (response.length < minimumLength) {
    throw new Error('Tracker response is shorter than the protocol requires.');
  }
}
 
async function sendWithRetries(options) {
  let lastError;
 
  for (let attempt = 0; attempt < options.retries; attempt += 1) {
    const timeoutMs = 15_000 * (2 ** attempt);
 
    try {
      return await sendOnce({ ...options, timeoutMs });
    } catch (error) {
      lastError = error;
      if (error.code !== 'TRACKER_TIMEOUT') {
        throw error;
      }
    }
  }
 
  throw lastError;
}
 
function sendOnce({
  socket,
  trackerUrl,
  port,
  message,
  transactionId,
  expectedAction,
  timeoutMs,
}) {
  return new Promise((resolve, reject) => {
    let timer;
 
    const cleanup = () => {
      clearTimeout(timer);
      socket.off('message', onMessage);
      socket.off('error', onError);
    };
 
    const onError = (error) => {
      cleanup();
      reject(error);
    };
 
    const onMessage = (response) => {
      if (response.length < 8) return;
      if (response.readUInt32BE(4) !== transactionId) return;
 
      const action = response.readUInt32BE(0);
      if (action !== expectedAction && action !== ACTION_ERROR) return;
 
      cleanup();
      resolve(response);
    };
 
    socket.on('message', onMessage);
    socket.on('error', onError);
 
    timer = setTimeout(() => {
      cleanup();
      const error = new Error(
        `UDP tracker did not reply within ${timeoutMs / 1000} seconds.`,
      );
      error.code = 'TRACKER_TIMEOUT';
      reject(error);
    }, timeoutMs);
 
    socket.send(message, port, trackerUrl.hostname, (error) => {
      if (error) onError(error);
    });
  });
}

src/Queue.js

JavaScript
import {
  BLOCK_LENGTH,
  blockLength,
  blocksPerPiece,
} from './torrent-parser.js';
 
export class Queue {
  constructor(torrent) {
    this.torrent = torrent;
    this.items = [];
    this.queuedPieces = new Set();
    this.choked = true;
  }
 
  enqueuePiece(pieceIndex) {
    if (this.queuedPieces.has(pieceIndex)) return;
    this.queuedPieces.add(pieceIndex);
 
    const count = blocksPerPiece(this.torrent, pieceIndex);
 
    for (let blockIndex = 0; blockIndex < count; blockIndex += 1) {
      this.items.push({
        index: pieceIndex,
        begin: blockIndex * BLOCK_LENGTH,
        length: blockLength(this.torrent, pieceIndex, blockIndex),
      });
    }
  }
 
  dequeue() {
    return this.items.shift();
  }
 
  get length() {
    return this.items.length;
  }
}

src/Pieces.js

JavaScript
import {
  BLOCK_LENGTH,
  blocksPerPiece,
  numberOfPieces,
} from './torrent-parser.js';
 
export class Pieces {
  constructor(torrent) {
    const makeTable = () =>
      Array.from({ length: numberOfPieces(torrent) }, (_, pieceIndex) =>
        new Array(blocksPerPiece(torrent, pieceIndex)).fill(false),
      );
 
    this.requested = makeTable();
    this.received = makeTable();
  }
 
  markRequested(pieceBlock) {
    this.requested[pieceBlock.index][this.blockIndex(pieceBlock)] = true;
  }
 
  markReceived(pieceBlock) {
    const blockIndex = this.blockIndex(pieceBlock);
    this.requested[pieceBlock.index][blockIndex] = true;
    this.received[pieceBlock.index][blockIndex] = true;
  }
 
  markFailed(pieceBlock) {
    const blockIndex = this.blockIndex(pieceBlock);
 
    if (!this.received[pieceBlock.index][blockIndex]) {
      this.requested[pieceBlock.index][blockIndex] = false;
    }
  }
 
  needed(pieceBlock) {
    const blockIndex = this.blockIndex(pieceBlock);
    return !this.requested[pieceBlock.index][blockIndex];
  }
 
  isDone() {
    return this.received.every((blocks) => blocks.every(Boolean));
  }
 
  percentDone() {
    const receivedBlocks = this.received.flat().filter(Boolean).length;
    const totalBlocks = this.received.flat().length;
    return Math.floor((receivedBlocks / totalBlocks) * 100);
  }
 
  blockIndex(pieceBlock) {
    return pieceBlock.begin / BLOCK_LENGTH;
  }
}

src/download.js

JavaScript
import { Buffer } from 'node:buffer';
import fs from 'node:fs';
import net from 'node:net';
 
import {
  buildHandshake,
  buildInterested,
  buildRequest,
  parseHandshake,
  parseMessage,
} from './message.js';
import { Pieces } from './Pieces.js';
import { Queue } from './Queue.js';
import {
  infoHash,
  numberOfPieces,
  totalSizeBigInt,
} from './torrent-parser.js';
import { getPeers } from './tracker.js';
 
const MAX_PEERS = 30;
const MAX_MESSAGE_LENGTH = 2 * 1024 * 1024;
 
export async function download(torrent, outputPath) {
  const peers = await getPeers(torrent);
 
  if (peers.length === 0) {
    throw new Error('The tracker returned no peers.');
  }
 
  const pieces = new Pieces(torrent);
  const fileDescriptor = fs.openSync(outputPath, 'w');
  const sockets = new Set();
  let finished = false;
 
  const totalSize = totalSizeBigInt(torrent);
  if (totalSize <= BigInt(Number.MAX_SAFE_INTEGER)) {
    fs.ftruncateSync(fileDescriptor, Number(totalSize));
  }
 
  console.log(`Tracker returned ${peers.length} peer(s).`);
 
  try {
    await new Promise((resolve, reject) => {
      let openConnections = 0;
      let lastError;
 
      const finishDownload = () => {
        if (finished) return;
        finished = true;
        process.stdout.write('\nDownload complete.\n');
 
        for (const socket of sockets) {
          socket.destroy();
        }
 
        resolve();
      };
 
      const peerFinished = (error) => {
        openConnections -= 1;
        if (error) lastError = error;
 
        if (!finished && openConnections === 0) {
          reject(
            lastError ?? new Error('All peer connections closed too early.'),
          );
        }
      };
 
      for (const peer of peers.slice(0, MAX_PEERS)) {
        openConnections += 1;
        connectToPeer({
          peer,
          torrent,
          pieces,
          fileDescriptor,
          sockets,
          isFinished: () => finished,
          finishDownload,
          peerFinished,
        });
      }
    });
  } finally {
    for (const socket of sockets) {
      socket.destroy();
    }
    fs.closeSync(fileDescriptor);
  }
}
 
function connectToPeer(context) {
  const {
    peer,
    torrent,
    pieces,
    fileDescriptor,
    sockets,
    isFinished,
    finishDownload,
    peerFinished,
  } = context;
 
  const socket = net.createConnection({ host: peer.ip, port: peer.port });
  const queue = new Queue(torrent);
  let inFlight = null;
  let connectionError = null;
 
  sockets.add(socket);
  socket.setTimeout(30_000);
 
  socket.on('connect', () => {
    socket.write(buildHandshake(torrent));
  });
 
  socket.on('timeout', () => {
    socket.destroy(new Error('Peer connection timed out.'));
  });
 
  socket.on('error', (error) => {
    connectionError = error;
  });
 
  socket.on('close', () => {
    sockets.delete(socket);
    if (inFlight) pieces.markFailed(inFlight);
    peerFinished(connectionError);
  });
 
  onWholeMessage(socket, (message, isHandshakeMessage) => {
    if (isFinished()) return;
 
    if (isHandshakeMessage) {
      const handshake = parseHandshake(message);
 
      if (handshake.protocol !== 'BitTorrent protocol') {
        throw new Error('Peer replied with an unknown protocol handshake.');
      }
 
      if (!handshake.infoHash.equals(infoHash(torrent))) {
        throw new Error('Peer handshake contains the wrong info hash.');
      }
 
      socket.write(buildInterested());
      return;
    }
 
    const parsed = parseMessage(message);
 
    if (parsed.id === null) return; // keep-alive
 
    switch (parsed.id) {
      case 0: // choke
        queue.choked = true;
        break;
      case 1: // unchoke
        queue.choked = false;
        requestNextBlock();
        break;
      case 4: // have
        queue.enqueuePiece(parsed.payload.readUInt32BE(0));
        requestNextBlock();
        break;
      case 5: // bitfield
        addBitfieldToQueue(parsed.payload, torrent, queue);
        requestNextBlock();
        break;
      case 7: // piece
        handlePiece(parsed.payload);
        break;
      default:
        // This small client safely ignores messages it does not use.
        break;
    }
  });
 
  function requestNextBlock() {
    if (queue.choked || inFlight || isFinished()) return;
 
    while (queue.length > 0) {
      const pieceBlock = queue.dequeue();
 
      if (pieces.needed(pieceBlock)) {
        pieces.markRequested(pieceBlock);
        inFlight = pieceBlock;
        socket.write(buildRequest(pieceBlock));
        return;
      }
    }
  }
 
  function handlePiece(pieceResponse) {
    if (!inFlight) return;
 
    const matchesRequest =
      pieceResponse.index === inFlight.index &&
      pieceResponse.begin === inFlight.begin &&
      pieceResponse.block.length === inFlight.length;
 
    if (!matchesRequest) {
      socket.destroy(new Error('Peer returned a block we did not request.'));
      return;
    }
 
    const offset =
      pieceResponse.index * torrent.info['piece length'] + pieceResponse.begin;
 
    fs.writeSync(
      fileDescriptor,
      pieceResponse.block,
      0,
      pieceResponse.block.length,
      offset,
    );
 
    pieces.markReceived(pieceResponse);
    inFlight = null;
 
    process.stdout.write(`Progress: ${pieces.percentDone()}%\r`);
 
    if (pieces.isDone()) {
      finishDownload();
    } else {
      requestNextBlock();
    }
  }
}
 
function addBitfieldToQueue(bitfield, torrent, queue) {
  const pieceCount = numberOfPieces(torrent);
 
  bitfield.forEach((byte, byteIndex) => {
    for (let bitIndex = 0; bitIndex < 8; bitIndex += 1) {
      const pieceIndex = byteIndex * 8 + bitIndex;
      if (pieceIndex >= pieceCount) return;
 
      const mask = 1 << (7 - bitIndex);
      if ((byte & mask) !== 0) {
        queue.enqueuePiece(pieceIndex);
      }
    }
  });
}
 
function onWholeMessage(socket, callback) {
  let saved = Buffer.alloc(0);
  let waitingForHandshake = true;
 
  socket.on('data', (received) => {
    saved = Buffer.concat([saved, received]);
 
    try {
      while (true) {
        if (waitingForHandshake) {
          if (saved.length < 1) return;
 
          const handshakeLength = saved.readUInt8(0) + 49;
          if (saved.length < handshakeLength) return;
 
          const handshake = saved.subarray(0, handshakeLength);
          saved = saved.subarray(handshakeLength);
          waitingForHandshake = false;
          callback(handshake, true);
          continue;
        }
 
        if (saved.length < 4) return;
 
        const messageLength = saved.readUInt32BE(0) + 4;
 
        if (messageLength > MAX_MESSAGE_LENGTH) {
          throw new Error('Peer sent an unreasonably large message.');
        }
 
        if (saved.length < messageLength) return;
 
        const message = saved.subarray(0, messageLength);
        saved = saved.subarray(messageLength);
        callback(message, false);
      }
    } catch (error) {
      socket.destroy(error);
    }
  });
}
